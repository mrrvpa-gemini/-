────────────────────────────────────────
ELEVATOR PITCH:
────────────────────────────────────────
CyAI is a next-generation AI platform offering specialized AI models for different
domains: mathematics, science, cybersecurity education, programming, web development,
and general knowledge. Unlike generic chatbots, each CyAI model has a highly
optimized personality, knowledge base, and system prompt for its specific domain.

The platform features a clean, dark-mode professional UI, a tiered subscription
model (Free vs Pro), fully automated payment via Stripe, and a powerful owner
control panel exclusively accessible to rvpa.

────────────────────────────────────────
VISION STATEMENT:
────────────────────────────────────────
"To be the #1 specialized AI platform for students, developers, and security
enthusiasts — combining expert-level AI knowledge with an exceptional user
experience at an accessible price point."

────────────────────────────────────────
TARGET AUDIENCE:
────────────────────────────────────────
  1. Students (SMP, SMA, Universitas) — math, science, study help
  2. Developers & programmers — code generation, debugging, architecture
  3. Security learners — ethical hacking study, CTF prep, cert study
  4. Hobbyists — general AI chat, creative writing, Q&A
  5. Professionals — quick coding help, documentation, code review

────────────────────────────────────────
BUSINESS MODEL:
────────────────────────────────────────
  - FREE TIER    : Limited (30 msg/day), 3 models accessible
  - PRO TIER     : $2.00/month, unlimited messages, all models including CyCodeVil
  - Revenue      : Monthly recurring subscriptions via Stripe
  - Growth       : Word of mouth, SEO, social media

────────────────────────────────────────
UNIQUE VALUE PROPOSITIONS:
────────────────────────────────────────
  1. Specialized AI models (not just one generic chatbot)
  2. Extremely affordable Pro tier ($2/month)
  3. Professional, fast UI with real-time streaming
  4. Indonesian + English language support
  5. Owner has full control (ban, reward, prank users)
  6. Built-in jailbreak protection
  7. Transparent pricing & easy payment

import { GoogleGenerativeAI, HarmBlockThreshold, HarmCategory } from '@google/generative-ai';
import { MODEL_CONFIG, SYSTEM_PROMPTS, ModelSlug } from './prompts';

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);

// Safety settings — balanced for educational platform
const SAFETY_SETTINGS = [
  { category: HarmCategory.HARM_CATEGORY_HARASSMENT, threshold: HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE },
  { category: HarmCategory.HARM_CATEGORY_HATE_SPEECH, threshold: HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE },
  { category: HarmCategory.HARM_CATEGORY_SEXUALLY_EXPLICIT, threshold: HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE },
  { category: HarmCategory.HARM_CATEGORY_DANGEROUS_CONTENT, threshold: HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE },
];

export interface ChatMessage {
  role: 'user' | 'assistant';
  content: string;
}

// Main function: returns a streaming async iterable
export async function streamGeminiResponse(
  modelSlug: ModelSlug,
  messages: ChatMessage[]
) {
  const config = MODEL_CONFIG[modelSlug];
  const systemPrompt = SYSTEM_PROMPTS[modelSlug];

  const model = genAI.getGenerativeModel({
    model: config.geminiModel,
    systemInstruction: systemPrompt,
    safetySettings: SAFETY_SETTINGS,
    generationConfig: {
      temperature: modelSlug === 'cycodevil' ? 0.7 : 0.8,
      topK: 40,
      topP: 0.95,
      maxOutputTokens: modelSlug === 'cycodevil' ? 8192 : 4096,
    },
  });

  // Convert messages to Gemini format (history = all but last message)
  const history = messages.slice(0, -1).map(m => ({
    role: m.role === 'user' ? 'user' as const : 'model' as const,
    parts: [{ text: m.content }],
  }));

  const lastMessage = messages[messages.length - 1].content;

  const chat = model.startChat({ history });
  const result = await chat.sendMessageStream(lastMessage);

  return result.stream;
}

// Non-streaming version (for admin/classification use)
export async function getGeminiResponse(

  const config = MODEL_CONFIG[modelSlug];

    model: config.geminiModel,
    systemInstruction: systemPrompt,
  });

  const result = await model.generateContent(userMessage);
  return result.response.text();
}

// app/api/chat/route.ts
import { NextRequest } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { streamGeminiResponse, ChatMessage } from '@/lib/gemini';
import { MODEL_CONFIG, ModelSlug } from '@/lib/prompts';
import { checkAndIncrementUsage } from '@/lib/redis';
import { detectJailbreak } from '@/lib/jailbreak';
import { saveMessage } from '@/lib/prisma-helpers';
import { prisma } from '@/lib/prisma';

export async function POST(req: NextRequest) {
  try {
    // 1. Authenticate
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return Response.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const user = await prisma.user.findUnique({
      where: { id: session.user.id }
    });

    if (!user) return Response.json({ error: 'User not found' }, { status: 404 });
    if (user.isBanned) {
      return Response.json({
        error: 'BANNED',
        reason: user.banReason || 'Your account has been suspended.'
      }, { status: 403 });
    }

    const body = await req.json();
    const { modelSlug, messages, chatId } = body as {
      modelSlug: ModelSlug;
      messages: ChatMessage[];
      chatId?: string;
    };

    // 2. Validate model
    const modelConfig = MODEL_CONFIG[modelSlug];
    if (!modelConfig) {
      return Response.json({ error: 'Invalid model' }, { status: 400 });
    }

    // 3. Check PRO requirement
    if (modelConfig.tier === 'pro' && !user.isPro) {
      return Response.json({
        error: 'PRO_REQUIRED',
        message: 'CyCodeVil requires a PRO subscription. Upgrade for $2/month.'
      }, { status: 403 });
    }

    // 4. Check rate limit (free users only)
    if (!user.isPro && modelConfig.dailyLimit) {
      const allowed = await checkAndIncrementUsage(user.id, modelSlug);
      if (!allowed) {
        return Response.json({
          error: 'RATE_LIMIT',
          message: `Daily limit of ${modelConfig.dailyLimit} messages reached. Upgrade to PRO for unlimited access!`
        }, { status: 429 });
      }
    }

    // 5. Jailbreak detection on last user message
    const lastUserMessage = messages.filter(m => m.role === 'user').at(-1)?.content || '';
    const jailbreakResult = detectJailbreak(lastUserMessage);
    if (jailbreakResult.detected) {
      await logJailbreakAttempt(user.id, req, lastUserMessage, jailbreakResult.reason);
      return Response.json({
        error: 'JAILBREAK_DETECTED',
        reason: jailbreakResult.reason,
        warning: 'Your message has been flagged as a policy violation.'
      }, { status: 400 });
    }

    // 6. Stream Gemini response
    const geminiStream = await streamGeminiResponse(modelSlug, messages);

    // 7. Create SSE stream
    let fullResponse = '';
    const readableStream = new ReadableStream({
      async start(controller) {
        try {
          for await (const chunk of geminiStream) {
            const text = chunk.text();
            fullResponse += text;
            const data = JSON.stringify({ type: 'chunk', text });
            controller.enqueue(new TextEncoder().encode(`data: ${data}\n\n`));
          }
          // Signal done
          controller.enqueue(new TextEncoder().encode(`data: ${JSON.stringify({ type: 'done' })}\n\n`));
          controller.close();

          // Save to DB asynchronously (don't await in stream)
          if (chatId) {
            saveMessage(chatId, 'assistant', fullResponse).catch(console.error);
          }
        } catch (error) {
          console.error('Gemini stream error:', error);
          controller.enqueue(new TextEncoder().encode(
            `data: ${JSON.stringify({ type: 'error', message: 'AI response failed' })}\n\n`
          ));
          controller.close();
        }
      },
    });

    return new Response(readableStream, {
      headers: {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache, no-transform',
        'Connection': 'keep-alive',
        'X-Accel-Buffering': 'no',
      },
    });

  } catch (error) {
    console.error('Chat API error:', error);
    return Response.json({ error: 'Internal server error' }, { status: 500 });
  }
}

async function logJailbreakAttempt(
  userId: string,
  req: NextRequest,
  content: string,

) {
  const ip = req.headers.get('x-forwarded-for') || req.headers.get('x-real-ip') || 'unknown';

    data: {
      userId,
      ip,
      content: content.substring(0, 1000), // limit stored content
      reason,
      action: 'warned',
    }
  });

'use client';

import { useState, useCallback, useRef } from 'react';
import { ModelSlug } from '@/lib/prompts';

export interface Message {
  id: string;
  role: 'user' | 'assistant';
  content: string;
  createdAt: Date;
}

export function useChat({ modelSlug, initialChatId }: {
  modelSlug: ModelSlug;
  initialChatId?: string;
}) {
  const [messages, setMessages] = useState<Message[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [usageCount, setUsageCount] = useState(0);
  const [dailyLimit] = useState(30);
  const chatIdRef = useRef<string | undefined>(initialChatId);

  const sendMessage = useCallback(async (content: string) => {
    const userMessage: Message = {
      id: crypto.randomUUID(),
      role: 'user',
      content,
      createdAt: new Date(),
    };

    const updatedMessages = [...messages, userMessage];
    setMessages(updatedMessages);
    setIsLoading(true);

    // Add empty AI message placeholder for streaming
    const aiMessageId = crypto.randomUUID();
    setMessages(prev => [...prev, {
      id: aiMessageId,
      role: 'assistant',
      content: '',
      createdAt: new Date(),
    }]);

    try {
      const response = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          modelSlug,
          messages: updatedMessages.map(m => ({
            role: m.role,
            content: m.content,
          })),
          chatId: chatIdRef.current,
        }),
      });

      if (!response.ok) {
        const error = await response.json();
        throw { code: error.error, message: error.message };
      }

      // Read SSE stream
      const reader = response.body!.getReader();
      const decoder = new TextDecoder();
      let aiContent = '';

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        const text = decoder.decode(value);
        const lines = text.split('\n');

        for (const line of lines) {
          if (!line.startsWith('data: ')) continue;
          const data = line.slice(6);
          try {
            const parsed = JSON.parse(data);
            if (parsed.type === 'chunk') {
              aiContent += parsed.text;
              // Update the AI message content in real-time
              setMessages(prev =>
                prev.map(m =>
                  m.id === aiMessageId ? { ...m, content: aiContent } : m
                )
              );
            } else if (parsed.type === 'done') {
              break;
            }
          } catch {}
        }
      }

      setUsageCount(prev => prev + 1);
    } catch (error) {
      // Remove empty AI message on error
      setMessages(prev => prev.filter(m => m.id !== aiMessageId));
      throw error;
    } finally {
      setIsLoading(false);
    }
  }, [messages, modelSlug]);

  const clearMessages = useCallback(() => {
    setMessages([]);
    chatIdRef.current = undefined;
  }, []);

  return { messages, isLoading, sendMessage, clearMessages, usageCount, dailyLimit };
}

import { NextAuthOptions } from 'next-auth';
import CredentialsProvider from 'next-auth/providers/credentials';
import { PrismaAdapter } from '@auth/prisma-adapter';
import bcrypt from 'bcryptjs';
import { prisma } from './prisma';
import { z } from 'zod';

const loginSchema = z.object({
  email: z.string().email(),
  password: z.string().min(8),
});

export const authOptions: NextAuthOptions = {
  adapter: PrismaAdapter(prisma) as any,
  session: { strategy: 'jwt', maxAge: 30 * 24 * 60 * 60 }, // 30 days
  pages: {
    signIn: '/login',
    error: '/login',
  },
  providers: [
    CredentialsProvider({
      name: 'Credentials',
      credentials: {
        email: { label: 'Email', type: 'email' },
        password: { label: 'Password', type: 'password' },
      },
      async authorize(credentials) {
        const parsed = loginSchema.safeParse(credentials);
        if (!parsed.success) throw new Error('Invalid credentials');

        const user = await prisma.user.findUnique({
          where: { email: parsed.data.email },
        });

        if (!user) throw new Error('No account found with this email.');
        if (user.isBanned) throw new Error(`BANNED:${user.banReason || 'Account suspended.'}`);
        if (!user.emailVerified) throw new Error('Please verify your email first.');

        const valid = await bcrypt.compare(parsed.data.password, user.password);
        if (!valid) throw new Error('Incorrect password.');

        // Update last active
        await prisma.user.update({
          where: { id: user.id },
          data: { lastActiveAt: new Date() },
        });

        return {
          id: user.id,
          email: user.email,
          username: user.username,
          displayName: user.displayName || user.username,
          avatar: user.avatar,
          isOwner: user.isOwner,
          isPro: user.isPro && (!user.proExpiresAt || user.proExpiresAt > new Date()),
          role: user.role,
        };
      },
    }),
  ],
  callbacks: {
    async jwt({ token, user, trigger, session }) {
      if (user) {
        token.id = user.id;
        token.username = (user as any).username;
        token.isOwner = (user as any).isOwner;
        token.isPro = (user as any).isPro;
        token.role = (user as any).role;
      }
      // Allow manual session update (e.g., after payment)
      if (trigger === 'update' && session) {
        token.isPro = session.isPro;
      }
      return token;
    },
    async session({ session, token }) {
      if (session.user) {
        session.user.id = token.id as string;
        session.user.username = token.username as string;
        session.user.isOwner = token.isOwner as boolean;
        session.user.isPro = token.isPro as boolean;
        session.user.role = token.role as string;
      }
      return session;
    },
  },
};

// Check usage and increment — returns true if allowed, false if exceeded
export async function checkAndIncrementUsage(
  userId: string,
  modelSlug: string,
  limit: number = 30
): Promise<boolean> {
  const today = new Date().toISOString().split('T')[0]; // "2024-01-15"
  const key = `usage:${userId}:${modelSlug}:${today}`;

  // Use INCR + EXPIRE pipeline for atomicity
  const pipeline = redis.pipeline();
  pipeline.incr(key);
  pipeline.ttl(key);
  const [count, ttl] = await pipeline.exec() as [number, number];

  // Set expiry if this is a new key (ttl = -1)
  if (ttl < 0) {
    await redis.expire(key, 86400); // 24 hours
  }

  return count <= limit;
}

// Get current usage count
export async function getUsageCount(userId: string, modelSlug: string): Promise<number> {
  const today = new Date().toISOString().split('T')[0];
  const key = `usage:${userId}:${modelSlug}:${today}`;
  const count = await redis.get<number>(key);
  return count || 0;
}

// Get total daily usage across all models
export async function getTotalDailyUsage(userId: string): Promise<number> {
  const today = new Date().toISOString().split('T')[0];
  const models = ['general', 'cyber', 'cycode', 'cycodevil'];
  const keys = models.map(m => `usage:${userId}:${m}:${today}`);
  const counts = await Promise.all(keys.map(k => redis.get<number>(k)));
  return counts.reduce((sum, c) => sum + (c || 0), 0);
}

export async function getOrCreateCustomer(email: string, userId: string): Promise<string> {

  const customer = await stripe.customers.create({
    email,
    metadata: { userId },
  });

}

import { NextRequest } from 'next/server';
import Stripe from 'stripe';
import { stripe } from '@/lib/stripe';
import { prisma } from '@/lib/prisma';
import { sendEmail } from '@/lib/email';
import { cacheUserProStatus } from '@/lib/redis';

export async function POST(req: NextRequest) {
  const body = await req.text();
  const sig = req.headers.get('stripe-signature')!;

  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(body, sig, process.env.STRIPE_WEBHOOK_SECRET!);
  } catch (err) {
    console.error('Webhook signature verification failed:', err);
    return new Response('Webhook Error', { status: 400 });
  }

  switch (event.type) {
    case 'checkout.session.completed': {
      const session = event.data.object as Stripe.Checkout.Session;
      const userId = session.metadata?.userId;
      if (!userId) break;

      const subscription = await stripe.subscriptions.retrieve(session.subscription as string);
      const periodEnd = new Date(subscription.current_period_end * 1000);

      await prisma.user.update({
        where: { id: userId },
        data: {
          isPro: true,
          proExpiresAt: periodEnd,
          proGrantedBy: 'stripe',
          stripeCustomerId: session.customer as string,
          stripeSubId: session.subscription as string,
        },
      });

      await prisma.transaction.create({
        data: {
          userId,
          amount: 2.00,
          currency: 'usd',
          status: 'paid',
          plan: 'pro_monthly',
          stripeId: session.id,
          stripeSubId: session.subscription as string,
          periodEnd,
        },
      });

      await cacheUserProStatus(userId, true);

      const user = await prisma.user.findUnique({
        where: { id: userId },
        select: { email: true, username: true }
      });
      if (user) {
        await sendEmail({
          to: user.email,
          subject: '⚡ Welcome to CyAI Pro!',
          template: 'pro_welcome',
          data: { username: user.username },
        });
      }
      break;
    }

    case 'customer.subscription.deleted': {
      const sub = event.data.object as Stripe.Subscription;
      const userId = sub.metadata?.userId;
      if (!userId) break;

      await prisma.user.update({
        where: { id: userId },
        data: {
          isPro: false,
          proExpiresAt: null,
          stripeSubId: null,
        },
      });

      await cacheUserProStatus(userId, false);

      const user = await prisma.user.findUnique({
        where: { id: userId },
        select: { email: true, username: true }
      });
      if (user) {
        await sendEmail({
          to: user.email,
          subject: 'Your CyAI Pro subscription has ended',
          template: 'pro_cancelled',
          data: { username: user.username },
        });
      }
      break;
    }

    case 'invoice.payment_failed': {
      const invoice = event.data.object as Stripe.Invoice;
      // Stripe will retry and send dunning emails automatically
      // We just log it
      console.log('Payment failed for customer:', invoice.customer);
      break;
    }
  }

  return new Response('OK', { status: 200 });
}

export async function POST(req: NextRequest) {
  const session = await getServerSession(authOptions);

  await requireOwner(session.user.id); // throws if not owner

  // Prevent banning yourself (owner)
  if (userId === session.user.id) {
    return Response.json({ error: 'Cannot ban yourself' }, { status: 400 });
  }

  const user = await prisma.user.update({
    where: { id: userId },
    data: {
      isBanned: true,
      banReason: reason || 'Violation of Terms of Service',
      bannedAt: new Date(),
      bannedBy: session.user.id,
    },
    select: { id: true, username: true, email: true },
  });

  // Invalidate all active sessions for this user
  await invalidateUserSessions(userId);

  // Log the action
  await prisma.notification.create({
    data: {
      userId,
      type: 'warning',
      title: 'Account Suspended',
      message: reason || 'Your account has been suspended for violating our Terms of Service.',
    },
  });

  return Response.json({
    success: true,
    message: `User ${user.username} has been banned.`
  });
}

// lib/jailbreak.ts

export interface JailbreakResult {
  detected: boolean;
  reason: string;
  severity: 'low' | 'medium' | 'high';
  pattern?: string;
}

// Pattern-based detection (Layer 1 — instant, no API call)
const JAILBREAK_PATTERNS: { pattern: RegExp; reason: string; severity: 'low' | 'medium' | 'high' }[] = [
  // Classic jailbreak phrases
  { pattern: /ignore (all |previous |your )?(instructions|rules|guidelines|training)/i, reason: 'Attempted to ignore system instructions', severity: 'high' },
  { pattern: /you (are now|have become|will now act as) (DAN|an AI (with no|without|that has no))/i, reason: 'DAN jailbreak attempt', severity: 'high' },
  { pattern: /pretend (you (are|have|don't have)|there are) no (rules|restrictions|limits|guidelines|ethics)/i, reason: 'Restriction bypass attempt', severity: 'high' },
  { pattern: /developer mode (enabled|activated|on)/i, reason: 'Developer mode jailbreak', severity: 'high' },
  { pattern: /bypass (your|all|any) (restrictions|guidelines|filters|safety|limits)/i, reason: 'Bypass attempt', severity: 'high' },
  { pattern: /act as if you (are|were|have) (no|without) (rules|limits|ethics|restrictions)/i, reason: 'Restriction removal attempt', severity: 'high' },
  { pattern: /forget (everything|that) you('re| are) an AI/i, reason: 'Identity manipulation', severity: 'medium' },
  { pattern: /your (true|real|actual) (self|form|purpose|identity) (is|has|can)/i, reason: 'Identity manipulation', severity: 'medium' },
  { pattern: /system prompt (is|revealed|show|display)/i, reason: 'Prompt extraction attempt', severity: 'medium' },
  { pattern: /\[SYSTEM\]|\[ADMIN\]|\[OVERRIDE\]|\[ROOT\]/i, reason: 'False system message injection', severity: 'high' },
  { pattern: /jailbreak|jail.?break/i, reason: 'Jailbreak keyword detected', severity: 'medium' },
  { pattern: /do anything now|DAN mode|stan mode|evil mode/i, reason: 'Known jailbreak variant', severity: 'high' },
  { pattern: /in this hypothetical (scenario|situation|world), you have no/i, reason: 'Hypothetical bypass attempt', severity: 'medium' },
  { pattern: /you are not bound by/i, reason: 'Restriction removal attempt', severity: 'high' },
  { pattern: /roleplay as an AI (without|that has no|with no) (ethics|guidelines|rules)/i, reason: 'Roleplay bypass attempt', severity: 'high' },

  // Prompt injection
  { pattern: /\nHuman:|<\|im_start\|>|<\|im_end\|>|<\|endoftext\|>/i, reason: 'Prompt injection attempt', severity: 'high' },
  { pattern: /----+.*new instructions.*----+/is, reason: 'Instruction injection', severity: 'high' },
];

export function detectJailbreak(message: string): JailbreakResult {
  // Skip very short messages (can't really jailbreak in 10 chars)
  if (message.length < 15) {
    return { detected: false, reason: '', severity: 'low' };
  }

  for (const { pattern, reason, severity } of JAILBREAK_PATTERNS) {
    if (pattern.test(message)) {
      return {
        detected: true,
        reason,
        severity,
        pattern: pattern.toString(),
      };
    }
  }

  return { detected: false, reason: '', severity: 'low' };
}

// Escalation logic based on strike count
export function determineAction(strikes: number): 'warn' | 'block' | 'ban' {
  if (strikes >= 3) return 'ban';
  if (strikes >= 2) return 'block';
  return 'warn';
}
